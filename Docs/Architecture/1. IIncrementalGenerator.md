### ⚙️ 1. IIncrementalGenerator (ENTRY POINT)

📄 FlowMapperGenerator.cs

```csharp
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp.Syntax;

namespace FlowMapper.SourceGenerator;

[Generator]
public class FlowMapperGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // 🔎 1. Syntax provider (filtra atributos Map<,>)
        var mappings = context.SyntaxProvider
            .CreateSyntaxProvider(
                predicate: IsMapperClass,
                transform: GetSemanticModel)
            .Where(x => x is not null);

        // 🧠 2. Pipeline principal
        var pipeline = mappings.Collect()
            .Select((items, _) => FlowPipeline.Execute(items!));

        // 🧾 3. Output generator
        context.RegisterSourceOutput(pipeline, EmitSource);
    }

    // 🔎 filtro leve (SEM semantic ainda)
    private static bool IsMapperClass(SyntaxNode node, CancellationToken ct)
    {
        return node is ClassDeclarationSyntax cls
            && cls.AttributeLists.Count > 0
            && cls.Modifiers.ToString().Contains("partial");
    }

    // 🧠 resolve semantic model
    private static MappingCandidate? GetSemanticModel(
        GeneratorSyntaxContext context,
        CancellationToken ct)
    {
        var classDecl = (ClassDeclarationSyntax)context.Node;

        var symbol = context.SemanticModel
            .GetDeclaredSymbol(classDecl, ct) as INamedTypeSymbol;

        if (symbol == null)
            return null;

        var attribute = symbol
            .GetAttributes()
            .FirstOrDefault(a => a.AttributeClass?.Name == "MapAttribute");

        if (attribute == null)
            return null;

        return MappingCandidateFactory.Create(symbol, attribute);
    }

    // 🧾 geração final
    private static void EmitSource(
        SourceProductionContext context,
        FlowModel model)
    {
        var code = FlowCodeGenerator.Generate(model);

        context.AddSource($"{model.MapperName}.g.cs", code);
    }
}
```

```csharp
using FlowMapper.Core.Domain;
using FlowMapper.SourceGenerator.Models;

namespace FlowMapper.SourceGenerator.Pipeline;

public static class FlowPipeline
{
    public static FlowModel Execute(IReadOnlyList<MappingCandidate> candidates)
    {
        var flows = new List<Flow>();

        foreach (var candidate in candidates)
        {
            var flow = FlowBuilder.Build(candidate);
            flows.Add(flow);
        }

        return new FlowModel(flows);
    }
}
```

### 🧱 5. FLOW BUILDER (CONVENTION ENGINE v1)

📄 FlowBuilder.cs

```csharp
using FlowMapper.Core.Domain;
using FlowMapper.SourceGenerator.Models;
using Microsoft.CodeAnalysis;

namespace FlowMapper.SourceGenerator.Pipeline.Builder;

public static class FlowBuilder
{
    public static Flow Build(MappingCandidate candidate)
    {
        var flow = new Flow
        {
            SourceType = candidate.SourceType.Name,
            DestinationType = candidate.DestinationType.Name,
            Policy = new MappingPolicy()
        };

        var sourceProps = candidate.SourceType
            .GetMembers()
            .OfType<IPropertySymbol>();

        var destProps = candidate.DestinationType
            .GetMembers()
            .OfType<IPropertySymbol>()
            .ToDictionary(x => x.Name);

        foreach (var sp in sourceProps)
        {
            if (destProps.TryGetValue(sp.Name, out var dp))
            {
                if (SymbolEqualityComparer.Default.Equals(sp.Type, dp.Type))
                {
                    flow.Properties.Add(new PropertyFlow
                    {
                        SourceProperty = sp.Name,
                        DestinationProperty = dp.Name
                    });
                }
            }
        }

        return flow;
    }
}
```

### 🧾 6. CODE GENERATOR (SIMPLIFICADO)

📄 FlowCodeGenerator.cs

```csharp
using FlowMapper.Core.Domain;

namespace FlowMapper.SourceGenerator.Pipeline.Generator;

public static class FlowCodeGenerator
{
    public static string Generate(FlowModel model)
    {
        var flow = model.Flows.First();

        var props = string.Join("\n",
            flow.Properties.Select(p =>
                $"{p.DestinationProperty} = source.{p.SourceProperty},"));
        return $@"
				public partial class {flow.SourceType}Mapper
				{
				    public {flow.DestinationType} Map({flow.SourceType} source)
				    {
				        return new {flow.DestinationType}
				        {
										{props}
				        };
				    }
				}";
    }
}
```

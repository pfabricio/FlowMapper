{
    public string MapperName { get; set; }

    public string SourceType { get; set; }
    public string DestinationType { get; set; }

    public string Profile { get; set; }

    public List<string> MappedProperties { get; set; } = new();

    public bool UsesConstructor { get; set; }
    public bool UsesFlatten { get; set; }
}

### ⚙️ 7. DOC COLLECTOR (CORE ENGINE)

```csharp
public static class FlowMapperDocCollector
{
    public static DocumentationModel Collect(IEnumerable<FlowModel> flows)
    {
        var doc = new DocumentationModel();

        foreach (var flow in flows.Flows)
        {
            doc.Mappers.Add(new MapperDocModel
            {
                MapperName = $"{flow.SourceType}Mapper",
                SourceType = flow.SourceType,
                DestinationType = flow.DestinationType,
                Profile = flow.ProfileName,
                UsesConstructor = flow.Properties.Any(p => p.Strategy == Strategy.Constructor),
                UsesFlatten = flow.Properties.Any(p => p.Strategy == Strategy.Flatten),
                MappedProperties = flow.Properties
                    .Select(p => $"{p.SourceProperty} → {p.DestinationProperty}")
                    .ToList()
            });
        }

        return doc;
    }
}
```

### 📄 README BUILDER

```csharp
public static class ReadmeGenerator
{
    public static string Generate(DocumentationModel model)
    {
        var sb = new StringBuilder();

        sb.AppendLine("# 🚀 FlowMapper");
        sb.AppendLine();
        sb.AppendLine("Compile-time object mapping engine powered by Roslyn Source Generators.");
        sb.AppendLine();

        sb.AppendLine("## 🧩 Mappers");
        sb.AppendLine();

        foreach (var mapper in model.Mappers)
        {
            sb.AppendLine($"### {mapper.MapperName}");
            sb.AppendLine($"- Source: `{mapper.SourceType}`");
            sb.AppendLine($"- Destination: `{mapper.DestinationType}`");
            sb.AppendLine($"- Profile: `{mapper.Profile}`");
            sb.AppendLine();

            sb.AppendLine("#### 🔁 Mappings");
            foreach (var m in mapper.MappedProperties)
            {
                sb.AppendLine($"- {m}");
            }

            if (mapper.UsesFlatten)
                sb.AppendLine("\n⚡ Uses Flatten Mapping");

            if (mapper.UsesConstructor)
                sb.AppendLine("\n🏗 Uses Constructor Mapping");

            sb.AppendLine();
        }

        return sb.ToString();
    }
}
```

### 🧠 11. PROFILE DOC GENERATION

```csharp
public static class ProfileDocGenerator
{
    public static List<string> GenerateProfiles(DocumentationModel model)
    {
        return model.Profiles
            .Distinct()
            .Select(p => $"## Profile: {p}")
            .ToList();
    }
}
```
